cmake_minimum_required(VERSION 3.24)
project(SwarmCS VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(mavlink GIT_REPOSITORY https://github.com/mavlink/c_library_v2.git GIT_TAG a2d0ecf011a4aaa7840480e43f4e38d8c9e79d29)
FetchContent_MakeAvailable(mavlink)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Location Positioning)
qt_standard_project_setup(REQUIRES 6.8)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
qt_add_executable(SwarmCS src/main.cpp src/VehicleLink.cpp src/VehicleParameters.cpp src/VehicleGroups.cpp src/VehicleTakeoff.cpp src/ParameterMetadata.cpp src/VehicleLink.h src/ComputerLocation.cpp src/ComputerLocation.h)
set(PARAMETER_METADATA assets/parameters/copter.json assets/parameters/plane.json assets/parameters/px4.json)
qt_add_resources(SwarmCS parameter_reference PREFIX "/parameters" BASE "assets/parameters" FILES ${PARAMETER_METADATA})
if(WIN32)
    target_link_libraries(SwarmCS PRIVATE ole32 uuid)
endif()
target_include_directories(SwarmCS SYSTEM PRIVATE ${mavlink_SOURCE_DIR})
target_link_libraries(SwarmCS PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Location Qt6::Positioning)
qt_add_qml_module(SwarmCS URI SwarmCS VERSION 1.0 QML_FILES Main.qml FlightMap.qml VehicleTree.qml VehicleCard.qml VehicleIcon.qml ParameterEditor.qml)
include(GNUInstallDirs)
install(TARGETS SwarmCS BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(VehicleLinkTest tests/VehicleLinkTest.cpp src/VehicleLink.cpp src/VehicleParameters.cpp src/VehicleGroups.cpp src/VehicleTakeoff.cpp src/ParameterMetadata.cpp src/VehicleLink.h)
    qt_add_resources(VehicleLinkTest parameter_reference_test PREFIX "/parameters" BASE "assets/parameters" FILES ${PARAMETER_METADATA})
    target_include_directories(VehicleLinkTest SYSTEM PRIVATE ${mavlink_SOURCE_DIR})
    target_include_directories(VehicleLinkTest PRIVATE src)
    target_link_libraries(VehicleLinkTest PRIVATE Qt6::Test Qt6::Network)
    add_test(NAME VehicleLink COMMAND VehicleLinkTest)
endif()
